cmake_minimum_required(VERSION 3.22.1)
project(focus_ai LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
get_filename_component(FOCUS_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
set(FETCHCONTENT_BASE_DIR "${FOCUS_ROOT}/.native/deps" CACHE PATH "Pinned native dependency cache")
include(FetchContent)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL ON CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS ON CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
if(EXISTS "${FOCUS_ROOT}/.native/llama.cpp/CMakeLists.txt")
  set(FETCHCONTENT_SOURCE_DIR_FOCUS_LLAMA "${FOCUS_ROOT}/.native/llama.cpp")
endif()
FetchContent_Declare(focus_llama
  URL "https://codeload.github.com/ggml-org/llama.cpp/tar.gz/839a572a4cdecdb80d211297b8e7d7323a6e99d2"
  URL_HASH SHA256=b78daf4ffc672d47d1631dfe613ada151e3bef42fe1c12bbe90374c46275c405)
FetchContent_MakeAvailable(focus_llama)
add_library(focus_ai SHARED focus_ai.cpp)
target_include_directories(focus_ai PRIVATE "${focus_llama_SOURCE_DIR}/include" "${focus_llama_SOURCE_DIR}/ggml/include")
target_link_libraries(focus_ai PRIVATE llama ggml android log)
